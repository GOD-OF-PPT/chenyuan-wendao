import {test,expect,type Page,type TestInfo} from '@playwright/test';

test.use({viewport:{width:1440,height:900},actionTimeout:8000,navigationTimeout:15000});
async function readonly(page:Page,work:()=>Promise<void>){const before=await recoveryDatabase(page);await work();expect(await recoveryDatabase(page)).toEqual(before);}
async function online(page:Page){const a=await activity(page);await expect(page.locator(`[data-testid="training-activity"][data-generation="${a.generation}"][data-running="true"]`)).toBeVisible();await expect(page.getByTestId('training-countdown')).toHaveText((a.cycleMs/1000).toFixed(1)+' 秒');return a;}
async function proof(page:Page,g:any,a:any){const raw=await recoveryDatabase(page),formal=resolve(g,a.nextCommand),rows=raw.rows['full-receipts'].filter((r:any)=>r.value.command?.commandId===a.nextCommand.commandId);expect(rows).toHaveLength(1);const r=rows[0].value;expect(stored(raw,'current')).toEqual(formal.nextState);expect(r.command).toEqual(a.nextCommand);expect(r.result).toEqual(formal.result);const now=r.proof?.resolvedAt??await page.evaluate(()=>Date.now());if(a.version>=5){expect(r.proof.activityBefore).toEqual(a);expect(r.proof.stateAfter).toEqual(formal.nextState);}else expect(r.proof).toBeUndefined();expect(now).toBeGreaterThanOrEqual(a.dueAt);expect(r.activity).toEqual(completeActivity(formal.nextState,a,formal.result,{commandId:a.nextCommand.commandId,now}));expect(stored(raw,'activity')).toEqual(r.activity);return raw;}
async function frame(page:Page,info:TestInfo,name:string){
 const rows=await page.locator('[data-testid="trial-goal"] :is(h2,p,small,strong,button):visible').evaluateAll(nodes=>nodes.map(n=>{const r=n.getBoundingClientRect(),nav=document.getElementById('primary-navigation')!.getBoundingClientRect(),hit=document.elementFromPoint(r.x+r.width/2,r.y+r.height/2);return{text:n.textContent,tag:n.tagName,font:parseFloat(getComputedStyle(n).fontSize),height:r.height,inside:r.x>=0&&r.right<=innerWidth&&r.y>=0&&r.bottom<=nav.top,hit:hit===n||!!hit&&n.contains(hit),clipped:n.scrollHeight>n.clientHeight+1||n.scrollWidth>n.clientWidth+1};}));
 await info.attach(name+'-geometry',{body:JSON.stringify(rows),contentType:'application/json'});await page.screenshot({path:info.outputPath(name+'.png')});
 for(const r of rows){expect(r.font,r.text??'').toBeGreaterThanOrEqual(r.tag==='SMALL'?14:18);expect(r.inside,r.text??'').toBe(true);if(r.tag==='BUTTON'){expect(r.height).toBeGreaterThanOrEqual(44);expect(r.hit).toBe(true);}else expect(r.clipped,r.text??'').toBe(false);}expect(await page.evaluate(()=>document.documentElement.scrollHeight)).toBeLessThanOrEqual(900);
}
