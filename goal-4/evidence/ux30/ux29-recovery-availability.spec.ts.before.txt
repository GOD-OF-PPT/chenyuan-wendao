 if(a.version>=5){expect(receipt.version).toBe(2);expect(receipt.proof.activityBefore).toEqual(a);expect(receipt.proof.stateAfter).toEqual(formal.nextState);expect(receipt.proof.resolvedAt).toBeGreaterThanOrEqual(a.dueAt);now=receipt.proof.resolvedAt;}else{expect(receipt.version).toBe(1);expect(Object.keys(receipt).sort()).toEqual(['format','version','input','fingerprint','activity','command','result'].sort());expect(receipt).not.toHaveProperty('proof');now=receipt.activity.status==='running'?receipt.activity.startedAt:a.dueAt;expect(now).toBeGreaterThanOrEqual(a.dueAt);}
 expect(receipt.activity).toEqual(completeActivity(formal.nextState,a,formal.result,{commandId:a.nextCommand.commandId,now}));expect(stored(raw,'activity')).toEqual(receipt.activity);expect(()=>validateActivityReceipt(receipt,formal.nextState)).not.toThrow();expect(raw.rows['full-receipts'].filter((r:any)=>r.key!==key)).toEqual(accepted.rows['full-receipts']);let changed=['current','previous','activity'];

}
