from pathlib import Path
import subprocess,json,os,sys,hashlib

b=Path(__file__).resolve().parent
r=b/'source'
node=Path('/Users/gordon/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/bin/node')
env=os.environ.copy();env['PATH']=str(node.parent)+':'+env['PATH']
assert json.loads((b/'core-build-terminal.json').read_text())['exit']==0
commands=[
 ('ui-full-declared.json',r/'web/playwright.idle.config.ts'),
 ('native-declared.json',r/'web/playwright.idle-recovery.config.ts'),
 ('recovery-ui-declared.json',r/'web/playwright.idle-recovery-ui.config.ts'),
 ('normal-full-declared.json',r/'web/playwright.idle-journeys.config.ts'),
 ('normal-declared.json',b/'normal.config.ts'),
 ('preflight-declared.json',b/'preflight.config.ts'),
 ('ui-all-declared.json',b/'ui-all.config.ts'),
 ('ui-declared.json',b/'ui.config.ts')]
def declared(report):
    rows=[]
    def walk(s):
        for spec in s.get('specs',[]):
            for test in spec['tests']:rows.append({'file':spec['file'],'title':spec['title'],'project':test['projectName']})
        for child in s.get('suites',[]):walk(child)
    for suite in report['suites']:walk(suite)
    return rows
result=[]
for name,config in commands:
    out=b/name;assert not out.exists()
    args=[str(node),str(r/'web/node_modules/@playwright/test/cli.js'),'test','--config',str(config),'--project=chromium','--list','--reporter=json']
    completed=subprocess.run(['rtk','proxy',*args],cwd=r/'web',env=env,capture_output=True)
    (b/(name+'.stderr.log')).write_bytes(completed.stderr)
    (b/(name+'.stdout.raw')).write_bytes(completed.stdout)
    assert completed.returncode==0,completed.stderr.decode()
    data=json.loads(completed.stdout);assert not data.get('errors')
    rows=declared(data);assert len(rows)==len({tuple(x.values()) for x in rows})
    out.write_bytes(completed.stdout)
    result.append({'name':name,'cases':len(rows),'files':len({x['file'] for x in rows}),'sha256':hashlib.sha256(completed.stdout).hexdigest()})
    if name=='ui-declared.json':
        scope=json.loads((r/'goal-4/evidence/check-cend10-12/scope.json').read_text());actual={'eligible':len(rows),'declared':rows,'wholeFiles':scope['wholeFiles'],'actualFiles':sorted({Path(x['file']).name for x in rows}),'pausedWholePrefixes':scope['plannedWholeMixedDeferrals']}
        assert actual['actualFiles']==sorted(scope['wholeFiles'])
        expected=json.loads((b/'expected-ui-union.json').read_text())
        key=lambda row:(Path(row['file']).name,row['title'],row['project'])
        assert sorted(map(key,rows))==sorted(map(key,expected['rows']))
        allRows=declared(json.loads((b/'ui-all-declared.json').read_text()))
        deferred=[row for row in allRows if any(row['title'].startswith(prefix) for prefix in scope['plannedWholeMixedDeferrals'])]
        assert sorted(map(key,[row for row in allRows if row not in deferred]))==sorted(map(key,rows))
        actual['actualWholeMixedDeferralsBeforeExecution']=deferred
        actual['rawWholeDeclarations']=len(allRows)
        actual['allActualPriorUnionIdentitiesPreserved']=True
        (b/'ui-scope-actual.json').write_text(json.dumps(actual,ensure_ascii=False,indent=2)+'\n')
(b/'declaration-terminal.json').write_text(json.dumps({'exit':0,'actual':result},ensure_ascii=False,indent=2)+'\n')
print(json.dumps(result,ensure_ascii=False))
