def reporterValue(v):
    if isinstance(v,dict):
        if set(v)=={'regexpSource','regexpFlags'}:return {}
        return {k:reporterValue(x) for k,x in v.items()}
    if isinstance(v,list):return [reporterValue(x) for x in v]
    return v

def declarations(report):
    rows=[]
    def walk(s):
        for spec in s.get('specs',[]):
            for t in spec['tests']:rows.append((spec['file'],spec['title'],t['projectName']))
        for child in s.get('suites',[]):walk(child)
    for suite in report['suites']:walk(suite)
    return sorted(rows)
