import {test,expect}from'./source/web/node_modules/@playwright/test/index.mjs';
import{feedbackExamples}from'./source/tests/action-feedback-fixtures.mjs';
import{getLifeOverview,previewAction}from'./source/src/full/queries.ts';
import{importRecoveryFixture,recoveryDatabase as database,runRecoveryPlan}from'./source/web/e2e/idle-recovery-ui-support.ts';
import{captureFixedPng}from'./source/web/e2e/png-capture.ts';
test.use({viewport:{width:1440,height:900},actionTimeout:8000,navigationTimeout:15000});
test('CEND04 supplemental actual six-second zero-month travel preserves the last month and records cave guidance without ending life',async({page},info)=>{
 const errors:string[]=[];page.on('pageerror',e=>errors.push(e.message));const before=await importRecoveryFixture(page,feedbackExamples().actions.get('END_LIFE').before);expect(getLifeOverview(before).remainingMonths).toBe(1);const trip=await runRecoveryPlan(page,{kind:'travel',location:'cave',maxCycles:1});expect(trip.a.cycleMs).toBe(6000);expect(trip.game.core.world.location).toBe('cave');expect(trip.game.core.world.elapsedMonths).toBe(before.core.world.elapsedMonths);expect(getLifeOverview(trip.game).remainingMonths).toBe(1);const raw=await database(page);await page.getByTestId('tab-0').click();
 const cultivate=previewAction(trip.game,{type:'CULTIVATE',method:'natural',count:1}),home={hud:await page.getByTestId('game-hud').innerText(),content:await page.locator('#game-content').innerText(),visibleWarning:await page.getByTestId('lifespan-warning').isVisible(),goal:await page.getByTestId('hub-target').innerText()};await captureFixedPng(page,info.outputPath('last-month-cave-main.png'));await page.getByTestId('hub-goal').click();await expect(page.getByTestId('goal-panel')).toBeVisible();const goal=await page.getByTestId('goal-panel').innerText();await captureFixedPng(page,info.outputPath('last-month-cave-goal.png'));await page.getByTestId('hub-close').click();expect(await database(page)).toEqual(raw);expect(errors).toEqual([]);await info.attach('actual-last-month-cave-observation',{body:JSON.stringify({life:getLifeOverview(trip.game),cultivate,home,goal,trip:{activity:trip.a,game:trip.game},allReadingDBUnchanged:true,errors,scope:'One formal actual online6sec travel with virtual boundary and receipt verification; following UI only viewed; not human acceptance or all-route impossibility.'}),contentType:'application/json'});
});
